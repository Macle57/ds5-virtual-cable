# Runtime dependencies for the Bluetooth side (prototype/ds5bridge and
# emulator/ds5emu/bridge.py).
#
# The USB/IP emulator itself -- emulator/ds5emu, everything except bridge.py --
# needs NONE of these. It is standard library only, deliberately, which is why
# 132 of the 173 unit tests run on a bare Python. See CONTRIBUTING.md.
#
#   python -m venv prototype\.venv
#   prototype\.venv\Scripts\python.exe -m pip install -r requirements.txt

# Lower bounds are conservative; the versions everything was actually verified
# against are noted alongside.

hidapi>=0.14      # the `hid` module; talks to the controller. verified: 0.15.0
numpy>=1.26       # audio buffer maths.                        verified: 2.5.2
av>=13            # PyAV: Opus encode/decode + swresample, libopus bundled in
                  # the Windows wheels. opuslib was rejected -- it ships no
                  # DLL on Windows.                            verified: 18.1.0
